git clone - копирует репозиторий.
git status - показывает изменения 
git add - добавляет файлы
git commit - сохраняет изменения
git push - отправляет на сервер
git pull - забирает изменения
git branch - показывает ветки
git checkout - переключает ветку
git merge - объединяет ветки 
git log - показывает историю
